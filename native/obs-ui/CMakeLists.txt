cmake_minimum_required(VERSION 3.24)
project(obs_ui_prototype VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets)
add_library(obs-ui-components STATIC src/components.cpp)
target_include_directories(obs-ui-components PUBLIC include)
target_link_libraries(obs-ui-components PUBLIC Qt6::Widgets)
if(MSVC)
  target_compile_options(obs-ui-components PUBLIC /utf-8 /Zc:__cplusplus /permissive-)
endif()
add_executable(obs-ui-gallery src/gallery.cpp)
target_link_libraries(obs-ui-gallery PRIVATE obs-ui-components)
add_executable(obs-ui-tests tests/components-test.cpp)
target_link_libraries(obs-ui-tests PRIVATE obs-ui-components)
enable_testing()
add_test(NAME components COMMAND obs-ui-tests -platform minimal)
